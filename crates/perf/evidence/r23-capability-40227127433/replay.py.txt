import gzip, hashlib, json, shutil, subprocess, sys, time
from pathlib import Path
source=Path('/tmp/r23-producer-observer/crates/perf/evidence/r23-producer-40214115988')
out=Path('/tmp/r23-producer-observer/crates/perf/evidence/r23-capability-40227127433')
scratch=Path('/tmp/r23-capability-replay-40227127433')
scratch.mkdir()
replay=scratch/'capture';shutil.copytree(source,replay)
assessment=scratch/'assessment';assessment.mkdir()
manifest=json.loads((source/'raw-sha256.json').read_text()); verified=[]
for name,entry in manifest.items():
 p=source/name
 if p.exists():
  data=p.read_bytes()
 elif (source/(name+'.parts.json')).exists():
  parts=json.loads((source/(name+'.parts.json')).read_text())
  data=b''.join(gzip.decompress((source/part).read_bytes()) for part in parts['parts'])
  assert len(data)==parts['size'] and hashlib.sha256(data).hexdigest()==parts['sha256'],name
 else:
  data=gzip.decompress((source/(name+'.gz')).read_bytes())
 assert len(data)==entry['size'] and hashlib.sha256(data).hexdigest()==entry['sha256'], name
 verified.append(name)
(out/'raw-verification.json').write_text(json.dumps({'source':str(source),'verified_entries':len(verified),'failures':[]},indent=2)+'\n')
statuses={}
for name in ['extract','endpoints','assess','new-boundaries','controls-assess']:
 text=gzip.decompress((source/(name+'.py.txt.gz')).read_bytes()).decode()
 text=text.replace('/tmp/r23-work/crates/perf/evidence/r23-producer-40214115988',str(replay))
 script=scratch/(name+'.py');script.write_text(text)
 args=[sys.executable,str(script)]
 if name in ['assess','new-boundaries','controls-assess']:args.append(str(replay))
 if name=='assess':args.append(str(assessment))
 label='replay-'+name
 (out/(label+'.command')).write_text(' '.join(args)+'\n')
 start=time.monotonic()
 with (out/(label+'.stdout')).open('w') as stdout,(out/(label+'.stderr')).open('w') as stderr:
  result=subprocess.run(args,stdout=stdout,stderr=stderr,check=False)
 (out/(label+'.exit')).write_text(str(result.returncode)+'\n')
 statuses[name]={'exit':result.returncode,'elapsed_seconds':time.monotonic()-start}
 (out/'replay-statuses.json').write_text(json.dumps(statuses,indent=2)+'\n')
 assert result.returncode==0,(name,result.returncode)
compared=[]
for name,root in [('capture-integrity.json',replay),('physical-endpoints.json',replay),('assessment.json',assessment),('new-boundaries.json',replay),('control-assessment.json',replay)]:
 assert json.loads((source/name).read_text())==json.loads((root/name).read_text()),name
 compared.append(name)
for ref in ['historical','candidate']:
 for name,root in [(ref+'-instrumented-v3-marked-low-writes.csv.gz',replay),(ref+'-instrumented-v3-marked-low-physical-writes.csv.gz',replay),(ref+'-intervals.csv.gz',assessment),(ref+'-new-boundaries.csv.gz',replay)]:
  assert gzip.decompress((source/name).read_bytes())==gzip.decompress((root/name).read_bytes()),name
  compared.append(name)
(out/'replay-comparison.json').write_text(json.dumps({'equal_json_or_decompressed_csv':compared,'failures':[]},indent=2)+'\n')
print(json.dumps({'verified_raw_entries':len(verified),'compared_outputs':len(compared),'statuses':statuses},indent=2))
