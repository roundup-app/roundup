from r23_init import *
from overhead_monitor import observe
import hashlib
bins=Path('/tmp/r23-queue-bins')
sched='sched:sched_switch,sched:sched_waking,sched:sched_wakeup,sched:sched_process_fork,sched:sched_process_exec,sched:sched_process_exit'
syscalls=['syscalls:sys_'+phase+'_'+call for call in ['read','write','writev','recvfrom','sendto','recvmsg','sendmsg','openat','close','dup','dup2','dup3','connect','accept4'] for phase in ['enter','exit']]
run('controls-before',['bash','-c','date -u --iso-8601=ns; cat /proc/sys/kernel/random/boot_id; uptime; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
for ref in ['historical','candidate']:
 events=[f'r23p:{ref}_'+stage for stage in ['input_line','output_line','enqueue','input','pump','output_publish','output_next'] if (root/(ref+'-'+{'input_line':'source-input','output_line':'source-output','output_publish':'output_publish-source-add','output_next':'output_next-source-add'}.get(stage,stage+'-symbol-add')+'.exit')).exists() and (root/(ref+'-'+{'input_line':'source-input','output_line':'source-output','output_publish':'output_publish-source-add','output_next':'output_next-source-add'}.get(stage,stage+'-symbol-add')+'.exit')).read_text().strip()=='0']
 for probe in ['common','marked']:
  repeat=1;attempt=0
  while repeat<=3:
   attempt+=1;busy=False
   for state in ['off','scheduler','full']:
    name=f'{ref}-{probe}-{repeat}-attempt{attempt}-{state}'
    run(name+'-background',['bash','-c','date -u --iso-8601=ns; cat /proc/loadavg; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
    run(name+'-definitions',['sudo','-n','cat','/sys/kernel/tracing/uprobe_events'])
    base=[str(bins/(probe+'-perf')),'--runs','1','--calls','1000','--rupd',str(bins/(ref+'-instrumented-rupd')),'--rup',str(bins/(ref+'-rup')),'--budgets',str(root/'diagnostic-budgets.json'),'--out',str(root/(name+'.json'))]
    cmd=base
    if state!='off':
     cmd=['sudo','-n','perf','record','-a','--clockid','mono','-m','32M','-e',sched]
     for event in syscalls:cmd+=['-e',event,'--exclude-perf']
     if state=='full':
      for event in events:cmd+=['-e',event]
     cmd+=['-o',str(root/(name+'.data')),'--','sudo','-n','-u','runner']+base
    status=observe(name,cmd)
    if not (root/(name+'.json')).exists():raise RuntimeError((name,status,'missing report'))
    report=json.loads((root/(name+'.json')).read_text())
    busy=busy or bool(report.get('skipped'))
   if not busy:repeat+=1
   if attempt>=9:raise RuntimeError('nine attempts; busy cycles retained, not complete')
run('controls-after',['bash','-c','date -u --iso-8601=ns; cat /proc/sys/kernel/random/boot_id; uptime; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
run('events-remove',['sudo','-n','perf','probe','-d','r23p:*'])
(root/'hashes-after.json').write_text(json.dumps({f.name:hashlib.sha256(f.read_bytes()).hexdigest() for f in bins.iterdir() if f.is_file()},indent=2)+'\n')
