from pathlib import Path
import json,statistics,sys,gzip,hashlib,collections,re
root=Path(sys.argv[1]);out=Path(sys.argv[2]);out.mkdir(exist_ok=False)
def read(p):
 return json.loads(gzip.decompress(p.read_bytes()) if p.suffix=='.gz' else p.read_bytes())
assessment={};rows=[]
for ref in ['historical','candidate']:
 for probe in ['common','marked']:
  group=f'{ref}-{probe}';reports={}
  for repeat in range(1,4):
   for state in ['off','scheduler','full']:
    paths=sorted(root.glob(f'{group}-{repeat}-attempt*-{state}.json'))+sorted(root.glob(f'{group}-{repeat}-attempt*-{state}.json.gz'))
    qualified=[(p,read(p)) for p in paths if not read(p)['skipped']]
    p,r=qualified[-1];assert r['runs']==1 and r['cpus']==4 and len(r['loads'])==1 and r['loads'][0]/r['cpus']<=2
    reports[repeat,state]=(p.name.removesuffix('.gz'),r)
  result={'reports':{f'{n}-{s}':p for (n,s),(p,r) in reports.items()},'metrics':{}}
  for metric in reports[1,'off'][1]['metrics']:
   values={s:[reports[n,s][1]['metrics'][metric] for n in range(1,4)] for s in ['off','scheduler','full']}
   effects={}
   for a,b in [('off','scheduler'),('scheduler','full'),('off','full')]:
    effects[b+' minus '+a]={'differences':[y-x for x,y in zip(values[a],values[b])],'ratios':[y/x for x,y in zip(values[a],values[b])]}
   result['metrics'][metric]={'values':values,'range':{s:[min(v),max(v)] for s,v in values.items()},'paired_effects':effects}
   if metric in ['terminal_write_p95_ms','write_to_output_p95_ms']:
    f=lambda v:f'{min(v):.6f}–{max(v):.6f}'
    rows.append('| '+group+' | `'+metric+'` | '+' | '.join(f(values[s]) for s in ['off','scheduler','full'])+' | '+f(effects['scheduler minus off']['differences'])+' | '+f(effects['full minus scheduler']['differences'])+' |')
  assessment[group]=result
(out/'assessment.json').write_text(json.dumps(assessment,indent=2)+'\n')
(out/'table.md').write_text('| Pin/probe | Metric | Off range (ms) | Scheduler range (ms) | Full range (ms) | Scheduler − off (ms) | Full − scheduler (ms) |\n|---|---|---:|---:|---:|---:|---:|\n'+'\n'.join(rows)+'\n')
# Failures are all actual command exits, never inferred from missing reports.
failures=['| Command record | Exit |','|---|---:|']
for p in sorted(root.glob('*.exit')):
 if p.read_text().strip()!='0':failures.append(f'| [{p.stem}.command]({p.stem}.command) | {p.read_text().strip()} |')
(out/'failures.md').write_text('\n'.join(failures)+'\n')
