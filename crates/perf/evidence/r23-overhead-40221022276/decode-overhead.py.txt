from r23_init import *
import collections,re
for data in sorted(root.glob('*.data')):
 name=data.stem
 run(name+'-evlist',['sudo','-n','perf','evlist','-v','-i',str(data)])
 run(name+'-header',['sudo','-n','perf','report','--stdio','--header-only','-i',str(data)])
 run(name+'-events',['sudo','-n','perf','script','--show-lost-events','--ns','-i',str(data),'-F','comm,pid,tid,cpu,time,event,trace'])
 counts=collections.Counter();lost=[]
 for line in (root/(name+'-events.stdout')).open():
  match=re.search(r'\s((?:r23p|sched|syscalls):[\w:]+):\s',line)
  if match:counts[match[1]]+=1
  if 'LOST' in line:lost.append(line.strip())
 (root/(name+'-counts.json')).write_text(json.dumps({'events':counts,'lost_records':lost},indent=2)+'\n')
