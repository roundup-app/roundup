from pathlib import Path
import gzip,hashlib,json,concurrent.futures
root=Path('/tmp/r23-overhead/crates/perf/evidence/r23-overhead-40221022276')
keep={'lifetime.json','verified-hashes.json','hashes-after.json','restore.json','diagnostic-budgets.json','classes.json','archived-build-settings.json'}
def pack(p):
 if not p.is_file():return None
 if p.suffix=='.gz':raw=gzip.decompress(p.read_bytes());return p.name.removesuffix('.gz'),hashlib.sha256(raw).hexdigest()
 raw=p.read_bytes();sha=hashlib.sha256(raw).hexdigest()
 if p.suffix in ['.stdout','.stderr','.log','.jsonl','.data'] or (p.suffix=='.json' and not p.name.endswith('.timing.json') and p.name not in keep):
  p.with_name(p.name+'.gz').write_bytes(gzip.compress(raw,compresslevel=6,mtime=0));p.unlink()
 return p.name,sha
files=list(root.iterdir())
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 hashes=dict(x for x in pool.map(pack,files) if x)
(root/'raw-sha256.json.gz').write_bytes(gzip.compress((json.dumps(hashes,indent=2)+'\n').encode(),mtime=0))
