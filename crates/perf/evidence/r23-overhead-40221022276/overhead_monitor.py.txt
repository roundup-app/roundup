from r23_init import *
import threading
bins=Path("/tmp/r23-queue-bins")
def monitor(stop,path):
 seen={}
 with path.open('w') as out:
  while not stop.is_set():
   before=time.monotonic_ns();owned=set()
   for d in Path('/proc').iterdir():
    if not d.name.isdigit(): continue
    try:
     exe=os.readlink(d/'exe')
     if exe.startswith(str(bins)+'/'): owned.add(int(d.name))
    except (FileNotFoundError,PermissionError,ProcessLookupError): pass
   pending=list(owned)
   while pending:
    pid=pending.pop()
    try:
     for task in (Path('/proc')/str(pid)/'task').iterdir():
      for child in (task/'children').read_text().split():
       if int(child) not in owned: owned.add(int(child));pending.append(int(child))
    except (FileNotFoundError,PermissionError,ProcessLookupError): pass
   for pid in sorted(owned):
    p=Path('/proc')/str(pid)
    try:
     facts={'pid':pid,'exe':os.readlink(p/'exe'),'cmdline':(p/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace'),'status':(p/'status').read_text(),'fds':{f.name:os.readlink(f) for f in (p/'fd').iterdir()},'tasks':{t.name:(t/'comm').read_text().strip() for t in (p/'task').iterdir()}}
     # Volatile status counters may differ; keep snapshots rather than inferring an FD across runs.
     key=json.dumps({k:v for k,v in facts.items() if k!='status'},sort_keys=True)
     if seen.get(pid)!=key:
      out.write(json.dumps({'before_ns':before,'after_ns':time.monotonic_ns(),**facts})+'\n');out.flush();seen[pid]=key
    except (FileNotFoundError,PermissionError,ProcessLookupError): pass
   stop.wait(.02)
def observe(name,cmd):
 if (root/(name+'.exit')).exists(): return int((root/(name+'.exit')).read_text())
 stop=threading.Event();thread=threading.Thread(target=monitor,args=(stop,root/(name+'-proc.jsonl')));thread.start()
 try: return run(name,cmd,'/tmp/r23-overhead')
 finally: stop.set();thread.join()
