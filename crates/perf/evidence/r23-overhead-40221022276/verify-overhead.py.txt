from pathlib import Path
import gzip,json,hashlib,concurrent.futures,shutil
root=Path('/tmp/r23-overhead/crates/perf/evidence/r23-overhead-40221022276')
old=json.loads(gzip.decompress((root/'raw-sha256.json.gz').read_bytes()))
def verify(p):
 if p.name=='raw-sha256.json.gz':return None
 raw=gzip.decompress(p.read_bytes()) if p.suffix=='.gz' else p.read_bytes()
 name=p.name.removesuffix('.gz') if p.suffix=='.gz' else p.name
 digest=hashlib.sha256(raw).hexdigest()
 if p.suffix=='.gz' and name in old:assert digest==old[name],name
 return name,digest
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 hashes=dict(x for x in pool.map(verify,list(root.iterdir())) if x)
(root/'raw-sha256.json.gz').write_bytes(gzip.compress((json.dumps(hashes,indent=2)+'\n').encode(),mtime=0))
assert json.loads((root/'verified-hashes.json').read_text())==json.loads((root/'hashes-after.json').read_text())
print('Verified retained gzip CRCs and raw capture hashes; executable hashes unchanged')
