from r23_init import *
import gzip,collections,hashlib,shutil
loads=[];records=[];ceilings={};samples=[]
for p in sorted(root.glob('*-attempt*-*.json')):
 if any(p.stem.endswith('-'+x) for x in ['off','scheduler','full']):
  r=json.loads(p.read_text());loads+=r['loads'];assert not r['skipped'] and r['runs']==1
  samples.append({'report':p.name,'per_run':r['per_run']})
  assert int((root/(p.stem+'.exit')).read_text())==1
  ceilings[p.stem]=(root/(p.stem+'.stdout')).read_text()+(root/(p.stem+'.stderr')).read_text()
for p in sorted(root.glob('*-counts.json')):
 c=json.loads(p.read_text());stem=p.stem.removesuffix('-counts');ev=(root/(stem+'-evlist.stdout')).read_text();err=(root/(stem+'-events.stderr')).read_text()
 assert c['events'].get('sched:sched_switch',0)>0 and c['events'].get('syscalls:sys_enter_write',0)>0
 active=sorted(set(__import__('re').findall(r'(?:sched|syscalls|r23p):\w+',ev)))
 source={k:v for k,v in c['events'].items() if k.startswith('r23p:')}
 if stem.endswith('-full'):assert source and any('input' in x for x in source) and any('output' in x for x in source)
 else:assert not source and not any(x.startswith('r23p:') for x in active)
 records.append({'recording':stem,'active_events':active,'source_hits':source,'zero_hit_active_events':[x for x in active if x not in c['events']],'lost_records':c['lost_records'],'decoder_stderr':err,'record_stderr':(root/(stem+'.stderr')).read_text()})
assert len(records)==24 and len(samples)==36
(root/'capture-summary.json').write_text(json.dumps({'controls':len(samples),'recordings':len(records),'load_range':[min(loads),max(loads)],'max_load_per_cpu':max(loads)/4,'recordings_detail':records},indent=2)+'\n')
(root/'samples.json').write_text(json.dumps(samples,indent=2)+'\n')
(root/'ceiling-output.json').write_text(json.dumps(ceilings,indent=2)+'\n')
assert json.loads((root/'verified-hashes.json').read_text())==json.loads((root/'hashes-after.json').read_text())
run('boot-after',['cat','/proc/sys/kernel/random/boot_id'])
assert (root/'boot-after.stdout').read_text()==gzip.decompress((root/'boot-before.stdout.gz').read_bytes()).decode()
run('jobs-after',['gh','api','repos/roundup-app/roundup/actions/runs/40221022276/jobs'])
