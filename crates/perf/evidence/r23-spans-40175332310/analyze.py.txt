from pathlib import Path
import re,json,csv,statistics,math
root=Path('/tmp/r23-correlated-write-spans/crates/perf/evidence/r23-spans-40175332310')
import gzip
def read(path):
 return path.read_text() if path.exists() else gzip.decompress(path.with_name(path.name+'.gz').read_bytes()).decode()

instant=re.compile(r'(\w+)=Instant \{ tv_sec: (\d+), tv_nsec: (\d+) \}')
def times(line):return {k:int(s)*10**9+int(n) for k,s,n in instant.findall(line)}
def stats(xs):
 xs=sorted(xs)
 return {'n':len(xs),'median_ns':statistics.median(xs),'p95_ns':xs[math.ceil(len(xs)*.95)-1],'max_ns':max(xs)} if xs else {'n':0}
summary={}
for ref,suffix in [(r,s) for r in ['historical','candidate'] for s in ['trace','sched-only']]:
 base=ref+'-marked';name=base if suffix=='trace' else base+'-sched-only'
 lines=read(root/(base+'-'+suffix+'.stderr')).splitlines()
 requests=[(re.search(r'id=(\d+)',x).group(1),times(x)) for x in lines if x.startswith('R23 request')]
 turns=[x for x in lines if x.startswith('R23 loop=') and 'turn=' in x]
 assert len(turns)==2200 and len(requests)==2200,(name,len(turns),len(requests))
 wakes=[times(x)['wake'] for x in lines if x.startswith('R23 receiver') and 'data=YQ==' in x]
 assert len(wakes)==2200
 rows=[]
 for ordinal,(line,(rid,request)) in enumerate(zip(turns,requests)):
  t=times(line); phase=re.search(r'loop=(\w+)',line).group(1); turn=int(re.search(r'turn=(\d+)',line).group(1))
  assert t['start']<=request['start']<=request['complete']
  rows.append({'loop':phase,'turn':turn,'warmup':turn<100,'request_id':rid,'start_ns':t['start'],'client_request_start_ns':request['start'],'client_request_complete_ns':request['complete'],'echo_ns':wakes[ordinal],'loop_echo_completed_ns':t['echo']})
 with (root/(name+'-writes.csv')).open('w') as f:
  w=csv.DictWriter(f,fieldnames=rows[0]);w.writeheader();w.writerows(rows)
 adapters=[times(x) for x in lines if x.startswith('R23 adapter')]
 # Setup output is also retained separately; ordinal mapping is restricted to the 2200 echo bytes.
 echo_adapters=[times(x) for x in lines if x.startswith('R23 adapter') and '"data":"YQ=="' in x]
 assert len(echo_adapters)==2200,(name,len(echo_adapters))
 # Raw echo identity is Terminal id + one-unread-write ordinal, not a changed payload.
 for r,a in zip(rows,echo_adapters):
  assert r['start_ns']<=a['decode_start']<=r['echo_ns'],(r,a)
 stages={}
 for label,a,b in [('decode','decode_start','decoded'),('branch_to_extract','decoded','extraction_start'),('extract','extraction_start','extracted'),('deserialize','offset' if ref=='historical' else 'extracted','deserialized'),('broadcast','deserialized','broadcast'),('adapter_total','extraction_start','broadcast')]:
  stages[label]=stats([x[b]-x[a] for x in echo_adapters])
 if ref=='historical':stages['offset_insert']=stats([x['offset']-x['extracted'] for x in echo_adapters])
 summary[name]={'writes':len(rows),'echo_adapters':len(echo_adapters),'adapter_all_frames':len(adapters),'stages':stages,'loops':{phase:{'request':stats([r['client_request_complete_ns']-r['client_request_start_ns'] for r in rows if r['loop']==phase and not r['warmup']]),'echo':stats([r['echo_ns']-r['start_ns'] for r in rows if r['loop']==phase and not r['warmup']])} for phase in ['write_to_output','terminal_write']}}
(root/'marker-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary,indent=2))
