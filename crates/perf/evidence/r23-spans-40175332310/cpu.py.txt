from pathlib import Path
import re,json,statistics,math,bisect
root=Path('/tmp/r23-correlated-write-spans/crates/perf/evidence/r23-spans-40175332310')
import gzip
def read(path):
 return path.read_text() if path.exists() else gzip.decompress(path.with_name(path.name+'.gz').read_bytes()).decode()

def stats(xs):
 xs=sorted(xs);return {'n':len(xs),'median_ns':statistics.median(xs),'p95_ns':xs[math.ceil(len(xs)*.95)-1],'max_ns':max(xs)} if xs else {'n':0}
instant=re.compile(r'(\w+)=Instant \{ tv_sec: (\d+), tv_nsec: (\d+) \}')
summary={}
for ref in ['historical','candidate']:
 name=ref+'-marked-sched-only';intervals={};active={};counts={}
 for line in read(root/(name+'-scheduler.stdout')).splitlines():
  head=re.search(r'\[(\d+)\]\s+(\d+)\.(\d+):\s+(sched:\w+): (.*)',line)
  if not head:continue
  cpu,sec,frac,event,detail=head.groups();t=int(sec)*10**9+int(frac.ljust(9,'0'));counts[event]=counts.get(event,0)+1
  if event!='sched:sched_switch':continue
  m=re.search(r':(\d+) \[\d+\] (\S+) ==> .*:(\d+) \[\d+\]',detail)
  assert m,line
  prev,state,nxt=m.groups();prev=int(prev);nxt=int(nxt)
  if cpu in active:
   tid,start=active[cpu]
   assert tid==prev,(line,active[cpu])
   intervals.setdefault(tid,[]).append((start,t))
  active[cpu]=(nxt,t)
 for tid in intervals:intervals[tid].sort()
 rows=[]
 for line in read(root/(name+'.stderr')).splitlines():
  if not line.startswith('R23 adapter') or '"data":"YQ=="' not in line:continue
  ts={k:int(s)*10**9+int(n) for k,s,n in instant.findall(line)}
  tid=int(re.search(r'tid=\d+/task/(\d+)',line).group(1))
  start,end=ts['extraction_start'],ts['broadcast']
  spans=intervals[tid];starts=[a for a,b in spans];i=max(0,bisect.bisect_right(starts,start)-1)
  cpu=0
  while i<len(spans) and spans[i][0]<end:
   a,b=spans[i];cpu+=max(0,min(end,b)-max(start,a));i+=1
  ordinal=len(rows);phase='write_to_output' if ordinal<1100 else 'terminal_write';turn=ordinal%1100
  rows.append({'loop':phase,'turn':turn,'warmup':turn<100,'tid':tid,**ts,'wall_ns':end-start,'cpu_ns':cpu,'off_cpu_ns':end-start-cpu})
 assert len(rows)==2200,(name,len(rows))
 import csv
 with (root/(name+'-adapter.csv')).open('w') as f:
  w=csv.DictWriter(f,fieldnames=rows[0]);w.writeheader();w.writerows(rows)
 summary[ref]={'events':counts,'adapter_cpu':stats([r['cpu_ns'] for r in rows]),'adapter_wall':stats([r['wall_ns'] for r in rows]),'adapter_off_cpu':stats([r['off_cpu_ns'] for r in rows]),'off_cpu_spans':sum(r['off_cpu_ns']>0 for r in rows),'stages':{label:stats([r[b]-r[a] for r in rows]) for label,a,b in [('decode','decode_start','decoded'),('extraction','extraction_start','extracted'),('deserialization','offset' if ref=='historical' else 'extracted','deserialized'),('broadcast','deserialized','broadcast')]}}
 summary[ref]['by_loop']={phase:{'measured':{key:stats([r[key] for r in rows if r['loop']==phase and not r['warmup']]) for key in ['wall_ns','cpu_ns','off_cpu_ns']},'warmups':{key:stats([r[key] for r in rows if r['loop']==phase and r['warmup']]) for key in ['wall_ns','cpu_ns','off_cpu_ns']}} for phase in ['write_to_output','terminal_write']}
 if ref=='historical':summary[ref]['stages']['offset_insert']=stats([r['offset']-r['extracted'] for r in rows])
(root/'adapter-cpu-summary.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary,indent=2))
