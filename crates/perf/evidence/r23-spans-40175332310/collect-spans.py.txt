exec(open('/tmp/r23-setup.py').read().split("run('machine'")[0])
import json,shutil,gzip
budgets=json.loads(Path('/tmp/r23-candidate/crates/perf/budgets.json').read_text())
for b in budgets.values():
 if 'linux-boxd' in b.get('baseline',{}): b['baseline']['linux-9v45']=b['baseline']['linux-boxd']
budgets.pop('hook_loop_ms')
(root/'diagnostic-budgets.json').write_text(json.dumps(budgets,indent=2)+'\n')
(root/'classes.json').write_text(json.dumps({'linux-9v45':'DIAGNOSTIC ONLY: copied linux-boxd to collect profiles; NOT a baseline or calibration'})+'\n')
# Clock bridge bounds REALTIME - MONOTONIC. Paired reads bracket the realtime read.
def clock_bridge(name):
 rows=[]
 for _ in range(100):
  before=time.clock_gettime_ns(time.CLOCK_MONOTONIC)
  real=time.time_ns()
  after=time.clock_gettime_ns(time.CLOCK_MONOTONIC)
  rows.append({'mono_before':before,'real':real,'mono_after':after})
 (root/(name+'.clock.json')).write_text(json.dumps(rows)+'\n')
for ref in ['historical','candidate']:
 for probe in ['common','marked']:
  name=ref+'-'+probe
  base=['/tmp/r23-span-bins/'+probe+'-perf','--runs','1','--calls','1000','--rupd','/tmp/r23-span-bins/'+ref+'-rupd','--rup','/tmp/r23-span-bins/'+ref+'-rup','--budgets',str(root/'diagnostic-budgets.json'),'--out',str(root/(name+'-control.json'))]
  run(name+'-control',base,'/tmp/r23-span-probe')
  base[-1]=str(root/(name+'-sched-only.json'))
  run(name+'-sched-only',['sudo','-n','perf','record','-a','--clockid','mono','-m','16M','-e','sched:sched_switch,sched:sched_waking,sched:sched_wakeup','-o','/tmp/r23-'+name+'-sched-only.data','--','sudo','-n','-u','runner']+base,'/tmp/r23-span-probe')
  run(name+'-sched-only-scheduler',['sudo','-n','perf','script','--ns','-i','/tmp/r23-'+name+'-sched-only.data','-F','comm,pid,tid,cpu,time,event,trace'])
  with (root/(name+'-sched-only.data.gz')).open('wb') as out:
   p=subprocess.Popen(['sudo','-n','cat','/tmp/r23-'+name+'-sched-only.data'],stdout=subprocess.PIPE)
   with gzip.GzipFile(fileobj=out,mode='wb',mtime=0) as z: shutil.copyfileobj(p.stdout,z)
   assert p.wait()==0
  base[-1]=str(root/(name+'-trace.json'))
  clock_bridge(name+'-before')
  run(name+'-trace',['sudo','-n','perf','record','-a','--clockid','mono','-m','16M','-e','sched:sched_switch,sched:sched_waking,sched:sched_wakeup','-o','/tmp/r23-'+name+'.data','--','sudo','-n','-u','runner','strace','-f','-yy','--absolute-timestamps=unix,ns','--syscall-times=ns','-s','8192','-o',str(root/(name+'.strace')),'-e','trace=read,write,writev,recvfrom,sendto,recvmsg,sendmsg,epoll_wait,epoll_pwait,futex,clone,clone3,execve,prctl,openat,close,dup,dup2,dup3,connect,accept4']+base,'/tmp/r23-span-probe')
  clock_bridge(name+'-after')
  run(name+'-scheduler',['sudo','-n','perf','script','--ns','-i','/tmp/r23-'+name+'.data','-F','comm,pid,tid,cpu,time,event,trace'])
  with (root/(name+'.sched.data.gz')).open('wb') as out:
   p=subprocess.Popen(['sudo','-n','cat','/tmp/r23-'+name+'.data'],stdout=subprocess.PIPE)
   with gzip.GzipFile(fileobj=out,mode='wb',mtime=0) as z: shutil.copyfileobj(p.stdout,z)
   assert p.wait()==0
