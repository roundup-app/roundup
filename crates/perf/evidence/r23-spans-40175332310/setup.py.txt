from pathlib import Path
import subprocess,os,shlex,time,json
root=Path('/tmp/r23-correlated-write-spans/crates/perf/evidence/r23-spans-40175332310');root.mkdir(parents=True,exist_ok=True)
def run(name,cmd,cwd='/tmp/r23-correlated-write-spans'):
 (root/(name+'.command')).write_text('cwd: '+cwd+'\n$ '+shlex.join(cmd)+'\n')
 started=time.perf_counter()
 with (root/(name+'.stdout')).open('w') as out,(root/(name+'.stderr')).open('w') as err:
  p=subprocess.run(cmd,cwd=cwd,stdout=out,stderr=err)
 (root/(name+'.elapsed-seconds')).write_text(str(time.perf_counter()-started)+'\n')
 (root/(name+'.exit')).write_text(str(p.returncode)+'\n');print(name,p.returncode,flush=True);return p.returncode
run('machine',['bash','-c','cat /etc/os-release; uname -a; lscpu; nproc; cat /sys/devices/system/cpu/online; cat /proc/self/status; cat /proc/sys/kernel/random/boot_id; rustc -Vv; cargo -V; /bin/bash --version; readlink -f /bin/sh; perf --version; strace --version; bpftrace --version; cat /proc/sys/kernel/perf_event_paranoid'])
(root/'lifetime.json').write_text(json.dumps({k:os.environ.get(k) for k in ['GITHUB_RUN_ID','GITHUB_JOB','RUNNER_NAME','GITHUB_SHA']},indent=2)+'\n')
qualification=Path('/home/runner/work/roundup/roundup/crates/perf/evidence/r23-host-40170055163/qualification.txt').read_text().split('$ ',1)[1].split('\nR23 host identity qualified;',1)[0]
assert run('qualification',['bash','-c',qualification])==0

 assert run(ref+'-checkout',['git','worktree','add','--detach','/tmp/r23-'+ref,sha])==0
