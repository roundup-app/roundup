from pathlib import Path
import gzip,shutil,subprocess,json,hashlib
r=Path('/tmp/r23-correlated-write-spans/crates/perf/evidence/r23-spans-40175332310')
for n in ['analyze','cpu','report','compress']:
 shutil.copy2('/tmp/r23-'+n+'.py',r/(n+'.py.txt'))
# Retain every raw recording, including the successful availability preflight.
with (r/'scheduler-preflight.data.gz').open('wb') as out:
 p=subprocess.Popen(['sudo','-n','cat','/tmp/r23-sched-preflight.data'],stdout=subprocess.PIPE)
 with gzip.GzipFile(fileobj=out,mode='wb',mtime=0) as z:shutil.copyfileobj(p.stdout,z)
 assert p.wait()==0
keep={'qualification.stdout','qualification.stderr','machine.stderr','final-hashes.stdout','final-hashes.stderr','hashes.stdout','hashes.stderr','adapter-cpu-summary.json','input-hashes.json','build-settings.json','lifetime.json','diagnostic-budgets.json','classes.json'}
for p in sorted(r.iterdir()):
 if p.name in keep:continue
 if p.name in ['marker-summary.json','controls-summary.json'] or p.name.endswith('-example.txt') or p.suffix in ['.stdout','.stderr','.strace','.csv','.log'] or p.name.endswith('.clock.json') or (p.suffix=='.json' and any(p.name.endswith('-'+k+'.json') for k in ['control','trace','sched-only'])):
  with p.open('rb') as src,(r/(p.name+'.gz')).open('wb') as dst:
   with gzip.GzipFile(fileobj=dst,mode='wb',mtime=0) as z:shutil.copyfileobj(src,z)
  p.unlink()
(r/'checksums.json').write_text(json.dumps({p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(r.iterdir()) if p.name.endswith('.gz')},indent=2)+'\n')
