from pathlib import Path
import csv,gzip,re,json
root=Path('/tmp/r23-correlated-write-spans/crates/perf/evidence/r23-spans-40175332310')
summary={}
for ref,main,inp,reader,cat in [('historical',21395,22496,22495,22497),('candidate',28258,29368,29367,29369)]:
 name=ref+'-marked'
 with gzip.open(root/(name+'-writes.csv.gz'),'rt') as f:rows=list(csv.DictReader(f))
 row=next(x for x in rows if x['loop']=='write_to_output' and x['turn']=='100')
 start=int(row['client_request_start_ns']);end=int(row['client_request_complete_ns']);echo=int(row['echo_ns'])
 active={};cpu=0;retained=[];wakes=[];switches=[]
 with gzip.open(root/(name+'-scheduler.stdout.gz'),'rt') as f:
  for line in f:
   m=re.search(r'\[(\d+)\]\s+(\d+)\.(\d+):\s+(sched:\w+): (.*)',line)
   if not m:continue
   core,sec,frac,event,detail=m.groups();t=int(sec)*10**9+int(frac.ljust(9,'0'))
   if start<=t<=echo:retained.append(line)
   if event=='sched:sched_waking' and start<=t<=echo and re.search(r'pid='+str(inp)+r'\b',detail):wakes.append(t)
   if event!='sched:sched_switch':continue
   sw=re.search(r':(\d+) \[\d+\] (\S+) ==> .*:(\d+) \[\d+\]',detail);assert sw,line
   prev,state,nxt=sw.groups();prev=int(prev);nxt=int(nxt)
   if core in active:
    tid,began=active[core];assert tid==prev
    if tid==main:cpu+=max(0,min(t,end)-max(began,start))
   active[core]=(nxt,t)
   if start<=t<=echo and nxt==inp:switches.append(t)
 with gzip.GzipFile(filename=str(root/(name+'-example-scheduler.txt.gz')),mode='wb',mtime=0) as f:f.write(''.join(retained).encode())
 assert wakes and switches
 first_wake=wakes[0];first_switch=next(t for t in switches if t>=first_wake)
 summary[ref]={'probe_main_tid':main,'input_tid':inp,'reader_tid':reader,'cat_tid':cat,'loop':'write_to_output','turn':100,'request_id':row['request_id'],'client_request_wall_ns':end-start,'client_main_on_cpu_ns':cpu,'client_main_off_cpu_ns':end-start-cpu,'input_first_waking_ns':first_wake,'input_first_switch_in_ns':first_switch,'input_waking_to_switch_ns':first_switch-first_wake,'note':'Combined ptrace/scheduler diagnostic only: off-CPU includes tracer stops; not normal-path latency or an additive stage budget.'}
(root/'scheduling-example.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary,indent=2))
