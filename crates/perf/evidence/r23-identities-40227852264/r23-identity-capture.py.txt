from importlib.machinery import SourceFileLoader
m=SourceFileLoader('ident','/tmp/r23-identity-init.py').load_module();root,run=m.root,m.run
import json,hashlib,time,sys
sys.path.insert(0,'/tmp')
from overhead_monitor import observe
sched='sched:sched_switch,sched:sched_waking,sched:sched_wakeup,sched:sched_process_fork,sched:sched_process_exec,sched:sched_process_exit'
syscalls=['syscalls:sys_'+phase+'_'+call for call in ['read','write','writev','recvfrom','sendto','recvmsg','sendmsg','openat','close','dup','dup2','dup3','connect','accept4'] for phase in ['enter','exit']]
run('controls-before',['bash','-c','date -u --iso-8601=ns; cat /proc/sys/kernel/random/boot_id; cat /proc/loadavg; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
for ref in ['historical','candidate']:
 for probe in ['common','marked']:
  repeat=1;attempt=0
  while repeat<=3:
   attempt+=1;busy=False
   for state in ['off','on']:
    name=f'{ref}-{probe}-{repeat}-attempt{attempt}-{state}'
    run(name+'-background',['bash','-c','date -u --iso-8601=ns; cat /proc/loadavg; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
    base=['/tmp/r23-queue-bins/'+probe+'-perf','--runs','1','--calls','1000','--rupd','/tmp/r23-queue-bins/'+ref+'-instrumented-rupd','--rup','/tmp/r23-queue-bins/'+ref+'-rup','--budgets',str(root/'diagnostic-budgets.json'),'--out',str(root/(name+'.json'))]
    cmd=base
    if state=='on':
     cmd=['sudo','-n','perf','record','-a','--clockid','mono','-m','32M','-e',sched]
     for event in syscalls:cmd+=['-e',event,'--exclude-perf']
     for stage in ['local','local_return','remote','remote_return']+(['send','send_return'] if ref=='candidate' else []):cmd+=['-e','r23i:'+ref+'_'+stage]
     cmd+=['-o',str(root/(name+'.data')),'--','sudo','-n','-u','runner']+base
    status=observe(name,cmd)
    if not (root/(name+'.json')).exists():raise RuntimeError((name,status,'missing report'))
    report=json.loads((root/(name+'.json')).read_text());busy=busy or bool(report.get('skipped'))
   if not busy:repeat+=1
   if attempt>=9:raise RuntimeError('nine cycles; busy attempts retained')
run('controls-after',['bash','-c','date -u --iso-8601=ns; cat /proc/sys/kernel/random/boot_id; cat /proc/loadavg; ps -eo pid,ppid,comm,pcpu,pmem,args --sort=-pcpu'])
run('uprobe-profile',['sudo','-n','cat','/sys/kernel/tracing/uprobe_profile'])
run('boot-after',['cat','/proc/sys/kernel/random/boot_id'])
(root/'hashes-after.json').write_text(json.dumps({f.name:hashlib.sha256(f.read_bytes()).hexdigest() for f in m.Path('/tmp/r23-queue-bins').iterdir()},indent=2)+'\n')
