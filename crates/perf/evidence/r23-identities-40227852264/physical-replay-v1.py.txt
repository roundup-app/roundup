"""Reuse the producer assessors for this recording's physical endpoint joins.
argv: repository, identity evidence, NEW scratch, NEW output.
"""
import sys,gzip,subprocess
from pathlib import Path
repo,source,scratch,out=map(Path,sys.argv[1:]);scratch.mkdir();out.mkdir()
previous=repo/'crates/perf/evidence/r23-producer-40214115988'
for ref in ['historical','candidate']:
 old=ref+'-instrumented-v3-marked-low';new='v2-'+ref+'-marked-1-attempt1-on'
 for suffix in ['.startup-1.log','.stderr','-events.stdout','-proc.jsonl']:
  src=source/(new+suffix)
  data=src.read_bytes() if src.exists() else gzip.decompress(Path(str(src)+'.gz').read_bytes())
  if suffix=='-events.stdout':
   with gzip.open(scratch/(old+suffix+'.gz'),'wb') as f:f.write(data)
  else:(scratch/(old+suffix)).write_bytes(data)
for script in ['extract','endpoints']:
 text=gzip.decompress((previous/(script+'.py.txt.gz')).read_bytes()).decode().replace('/tmp/r23-work/crates/perf/evidence/r23-producer-40214115988',str(scratch))
 target=scratch/(script+'.py');target.write_text(text);subprocess.run([sys.executable,str(target)],check=True)
for script in ['assess','new-boundaries']:
 target=scratch/(script+'.py');target.write_bytes(gzip.decompress((previous/(script+'.py.txt.gz')).read_bytes()))
 if script=='assess':subprocess.run([sys.executable,str(target),str(scratch),str(out/'assessment')],check=True)
# Keep derived joins in output; names refer to the assessor's aliases, not old samples.
import shutil
for pattern in ['*-physical-writes.csv.gz','*-markers.jsonl.gz','physical-endpoints.json','capture-integrity.json']:
 for p in scratch.glob(pattern):shutil.copy2(p,out/p.name)
