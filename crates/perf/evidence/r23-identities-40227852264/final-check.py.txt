from pathlib import Path
import gzip,json,hashlib,shutil,subprocess
root=Path('/tmp/r23-native-identities/crates/perf/evidence/r23-identities-40227852264')
def read(name):
 p=root/name
 return p.read_bytes() if p.exists() else gzip.decompress(Path(str(p)+'.gz').read_bytes())
assert read('boot-before.stdout')==read('boot-after.stdout')==read('v2-boot-after.stdout')
assert json.loads(read('hashes-before.json'))==json.loads(read('hashes-after.json'))
x=json.loads(gzip.decompress((root/'native-assessment/identity-assessment.json.gz').read_bytes()))
for ref,d in x.items():
 assert d['task_join_count']==d['physical_write_count']==2200 and d['unsupported_count']==d['absent_count']==d['unmatched_entry_count']==0
 assert d['pointer_reuse_count']>0 and not d['reported_LOST_lines']
assert x['candidate']['send_join_count']==2200
with gzip.open(root/'native-assessment/candidate-joins.json.gz','rt') as f:s=json.load(f)['send']
assert [j['send']['entry']['offset'] for j in s]==list(range(6,2206))
assert all(j['send']['entry']['length']==1 and j['send']['entry']['byte']==97 for j in s)
# Compression follows all controls/assessment. Original bytes stay hash-verifiable.
for p in list(root.rglob('*')):
 if p.is_file() and p.suffix=='.json' and not p.name.endswith('.timing.json') and (p.stat().st_size>1000):
  with p.open('rb') as f,gzip.open(str(p)+'.gz','wb',compresslevel=6) as out:shutil.copyfileobj(f,out)
  p.unlink()
manifest={}
for p in root.rglob('*.gz'):
 h=hashlib.sha256();size=0
 with gzip.open(p,'rb') as f:
  while b:=f.read(1024*1024):h.update(b);size+=len(b)
 manifest[str(p.relative_to(root)).removesuffix('.gz')]={'sha256':h.hexdigest(),'size':size}
with gzip.open(root/'raw-sha256.json.gz','wt') as f:json.dump(manifest,f,indent=2)
print('PASS: boot, fixed hashes, 4400 physical/task joins, 2200 candidate send joins, pointer reuse, decoded loss and raw manifest')
