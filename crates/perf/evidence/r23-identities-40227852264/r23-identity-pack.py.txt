from pathlib import Path
import gzip,shutil,hashlib,json,concurrent.futures
root=Path('/tmp/r23-native-identities/crates/perf/evidence/r23-identities-40227852264')
def pack(p):
 digest=hashlib.sha256()
 with p.open('rb') as f,gzip.open(str(p)+'.gz','wb',compresslevel=1) as out:
  while chunk:=f.read(1024*1024):digest.update(chunk);out.write(chunk)
 size=p.stat().st_size;p.unlink();return p.name,{'sha256':digest.hexdigest(),'size':size}
files=[p for p in root.iterdir() if p.is_file() and p.suffix in ['.stdout','.stderr','.log','.data','.jsonl'] and p.stat().st_size>0]
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:manifest=dict(pool.map(pack,files))
(root/'packed-raw-sha256.json').write_text(json.dumps(manifest,indent=2)+'\n')
