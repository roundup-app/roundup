"""Offline assessment; argv: evidence, NEW output. Uses perf-decoded text only."""
from pathlib import Path
import sys,gzip,json,re,collections,statistics,csv
source,out,physical_root=map(Path,sys.argv[1:]);out.mkdir()
def read(name):
 p=source/name
 return p.read_text() if p.exists() else gzip.decompress(Path(str(p)+'.gz').read_bytes()).decode()
controls=[];table=['| Set / pin / observer | Metric | Off range (ms) | On range (ms) | Paired on/off range |','|---|---|---:|---:|---:|']
for version in ['', 'v2-']:
 for ref in ['historical','candidate']:
  for probe in ['common','marked']:
   pairs=[]
   for i in range(1,4):
    pair=[]
    for state in ['off','on']:
     name=f'{version}{ref}-{probe}-{i}-attempt{i}-{state}'
     r=json.loads(read(name+'.json'));status=int(read(name+'.exit'))
     assert max(r['loads'])/r['cpus']<=2 and not r['skipped'],name
     controls.append({'name':name,'status':status,'loads':r['loads'],'cpus':r['cpus'],'metrics':r['metrics'],'skipped':r['skipped']});pair.append(r)
    pairs.append(pair)
   for metric in ['terminal_write_p95_ms','write_to_output_p95_ms']:
    a=[x[0]['metrics'][metric] for x in pairs];b=[x[1]['metrics'][metric] for x in pairs];ratios=[y/x for x,y in zip(a,b)]
    table.append(f'| {version or "v1"}/{ref}/{probe} | {metric} | {min(a):.6f}–{max(a):.6f} | {min(b):.6f}–{max(b):.6f} | {min(ratios):.3f}–{max(ratios):.3f} |')
(out/'controls.json').write_text(json.dumps(controls,indent=2)+'\n');(out/'table.md').write_text('\n'.join(table)+'\n')
assessment={}
pattern=re.compile(r'^\s*.*?\s+(\d+)/(\d+)\s+\[\d+\]\s+(\d+)\.(\d{9}):\s+r23i:(\w+):\s+(.*)')
for ref in ['historical','candidate']:
 name='v2-'+ref+'-marked-1-attempt1-on';log=read(name+'.startup-1.log')
 meta=set(map(int,re.findall(r'target=Id\((\d+)\)',log)));events=[];counts=collections.Counter();offsets=collections.Counter();unsupported=[];pointers=collections.defaultdict(set);stacks=collections.defaultdict(list);bounds=[]
 for line in read(name+'-events.stdout').splitlines():
  m=pattern.match(line)
  if not m:continue
  pid,tid,sec,ns,kind,detail=m.groups();stamp=int(sec)*10**9+int(ns);pid=int(pid);tid=int(tid);counts[kind]+=1
  fields={k:int(v) for k,v in re.findall(r'(\w+)=(\d+)',detail)}
  record={'pid':pid,'tid':tid,'ns':stamp,'event':kind,**fields};events.append(record)
  if kind.endswith(('_local','_remote')):
   offset=fields.get('idoff');offsets[offset]+=1
   key='id'+str(offset)
   if key not in fields:unsupported.append(record);continue
   record['task_id']=fields[key];pointers[(pid,fields['task'])].add(fields[key]);record['in_taskmeta']=fields[key] in meta
  if kind.endswith('_return'):
   key=(pid,tid,kind.removesuffix('_return'))
   if stacks[key]:
    start=stacks[key].pop();bounds.append({'entry':start,'return_ns':stamp})
  else:stacks[(pid,tid,kind)].append(record)
 with gzip.open(out/(ref+'-native-events.jsonl.gz'),'wt') as f:
  for e in events:f.write(json.dumps(e)+'\n')
 native=[e for e in events if 'task_id' in e];missing=[e for e in native if not e['in_taskmeta']]
 reuse=[{'pid':pid,'pointer':ptr,'task_ids':sorted(ids)} for (pid,ptr),ids in pointers.items() if len(ids)>1]
 assessment[ref]={'name':name,'counts':dict(counts),'id_offsets':dict(offsets),'recovered':len(native),'unsupported':unsupported[:10],'unsupported_count':len(unsupported),'ids_absent_from_taskmeta':missing[:10],'absent_count':len(missing),'pointer_reuse_count':len(reuse),'pointer_reuse_examples':reuse[:5],'entry_return_pairs':len(bounds),'unmatched_entry_count':sum(map(len,stacks.values())),'reported_LOST_lines':[l for l in read(name+'-events.stdout').splitlines() if 'LOST' in l]}
 # This recording's reused producer assessor supplies physical/request ordinals.
 physical=physical_root/(ref+'-instrumented-v3-marked-low-physical-writes.csv.gz')
 if physical.exists():
  with gzip.open(physical,'rt') as f:rows=list(csv.DictReader(f))
  joins=[];send_joins=[]
  for row in rows:
   start=int(row['client_start_ns']);end=int(row['dispatch_enter_ns']);task=int(re.search(r'Id\((\d+)\)',row['dispatch_task'])[1]);pid=int(row['daemon_pid'])
   matching=[e for e in native if e['pid']==pid and e['task_id']==task and start<=e['ns']<=end]
   if matching:joins.append({'ordinal':int(row['ordinal']),'request_id':row['request_id'],'terminal':row['terminal'],'dispatch_task':task,'native':matching[-1],'physical_write_tid':row['physical_write_tid'],'physical_read_tid':row['physical_read_tid']})
   if ref=='candidate':
    matches=[b for b in bounds if b['entry']['event']=='candidate_send' and b['entry']['tid']==int(row['physical_read_tid']) and int(row['physical_read_exit_ns'])<=b['entry']['ns']<=int(row['output_forward_receive_ns'])]
    if len(matches)==1:send_joins.append({'ordinal':int(row['ordinal']),'physical_read_exit_ns':int(row['physical_read_exit_ns']),'output_receive_ns':int(row['output_forward_receive_ns']),'send':matches[0]})
  assessment[ref]['physical_write_count']=len(rows);assessment[ref]['task_join_count']=len(joins);assessment[ref]['task_join_examples']=joins[:3]
  assessment[ref]['send_join_count']=len(send_joins);assessment[ref]['send_join_examples']=send_joins[:3]
  with gzip.open(out/(ref+'-joins.json.gz'),'wt') as f:json.dump({'task':joins,'send':send_joins},f)
(out/'identity-assessment.json').write_text(json.dumps(assessment,indent=2)+'\n')
print(json.dumps({k:{x:v[x] for x in ['counts','id_offsets','recovered','unsupported_count','absent_count','pointer_reuse_count','task_join_count','send_join_count'] if x in v} for k,v in assessment.items()},indent=2))
