#!/bin/bash
# Run with bash; kept as .txt so it is evidence, not a launch site (loop/launch-sites.json).
# Reproduces the 9V45 diagnosis. Build the historical reference first:
#   git worktree add /tmp/hist 2029ed2c4 && (cd /tmp/hist && CARGO_TARGET_DIR=/tmp/hist-target cargo build --release -p rupd -p rup -p perf)
# and put a budgets.json and classes.json under /tmp/scr that copy linux-boxd into linux-9v45.
M=$PWD; H=/tmp/hist; HT=/tmp/hist-target/release
O=$M/crates/perf/evidence/linux-9v45/diagnosis
hist() { (cd $H && $HT/perf --runs 11 --rupd "$2" --out $O/$1.json 2> $O/$1.log; echo $? > $O/$1.exit); }
cur() { (cd $M && $M/target/release/perf --runs 11 --budgets /tmp/scr/budgets.json "${@:2}" --out $O/$1.json 2> $O/$1.log; echo $? > $O/$1.exit); }
for i in 1 2 3 4 5 6 7 8 9; do
  hist historical-$i $HT/rupd
  cur current-$i
done
for i in 1 2 3; do
  cur newprobe-histdaemon-$i --rupd $HT/rupd --rup $HT/rup
  hist oldprobe-curdaemon-$i $M/target/release/rupd
done
