from pathlib import Path
import subprocess, gzip, hashlib, json, re, collections, csv, shutil
root = Path('/tmp/roundup-r23')
old = root / 'crates/perf/evidence/r23-host-40170055163'
out = root / 'crates/perf/evidence/r23-attribution-40174335770'
scratch = Path('/tmp/r23-raw')
statuses = []
def run(name, argv):
    result = subprocess.run(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE, cwd=root)
    (out / (name + '.command')).write_text('cwd: ' + str(root) + '\n$ ' + __import__('shlex').join(argv) + '\n')
    for ext, data in [('stdout',result.stdout),('stderr',result.stderr)]:
        with gzip.GzipFile(filename=str(out / (name + '.' + ext + '.gz')),mode='wb',mtime=0) as stream:
            stream.write(data)
    (out / (name + '.exit')).write_text(str(result.returncode)+'\n')
    statuses.append({'command':name,'exit':result.returncode})
    return result.stdout.decode()
run('profiler-version',['perf','version'])
# Retain the rejected option separately; its nonzero status is not a profile result.
run('unsupported-option',['perf','script','-i',str(scratch/'historical.data'),'--no-call-graph'])
identities = {}
summary = {}
for ref in ['historical','candidate']:
    data = gzip.decompress((old / (ref + '-perf.data.gz')).read_bytes())
    path = scratch / (ref + '.data')
    path.write_bytes(data)
    identities[ref] = {'compressed_sha256':hashlib.sha256((old / (ref + '-perf.data.gz')).read_bytes()).hexdigest(),'raw_sha256':hashlib.sha256(data).hexdigest()}
    run(ref+'-header',['perf','script','-i',str(path),'--header-only'])
    run(ref+'-buildids',['perf','buildid-list','-i',str(path)])
    text = run(ref+'-events',['perf','script','-i',str(path),'--show-task-events','--show-mmap-events','--show-switch-events','-G','-F','comm,pid,tid,time,event,ip,sym,dso'])
    records = []
    for line in text.splitlines():
        m = re.match(r'\s*(\S+)\s+(\d+)/(\d+)\s+([\d.]+): (.*)',line)
        if m:
            comm,pid,tid,time,detail = m.groups()
            records.append((comm,int(pid),int(tid),float(time),detail))
    daemon = next(r[1] for r in records if f'PERF_RECORD_COMM exec: {ref}-rupd:' in r[4])
    probe = next(r[1] for r in records if 'PERF_RECORD_COMM exec: compatible-perf:' in r[4])
    start = next(r[3] for r in records if r[1] == daemon and r[4].startswith('PERF_RECORD_COMM: terminal-11:'))
    end = min(r[3] for r in records if r[1] == daemon and r[4].startswith('PERF_RECORD_EXIT'))
    samples = [r for r in records if r[4].startswith('cpu-clock:')]
    counts = collections.Counter()
    for comm,pid,tid,time,detail in samples:
        window = 'echo-terminal-to-teardown' if start <= time < end else 'outside'
        role = 'probe' if pid == probe else 'daemon' if pid == daemon else 'child-or-other'
        counts[window,role,pid,tid,comm] += 1
    with (out/(ref+'-threads.csv')).open('w') as f:
        w=csv.writer(f); w.writerow(['window','role','pid','tid','comm','cpu_clock_samples'])
        for key,count in sorted(counts.items()): w.writerow([*key,count])
    grouped = collections.Counter()
    for key,count in counts.items(): grouped[key[0],key[1]] += count
    relevant = [r for r in records if r[4].startswith('PERF_RECORD_COMM') and (r[1] in (daemon,probe) or 'exec: cat:' in r[4]) or r[1]==daemon and r[4].startswith('PERF_RECORD_EXIT')]
    (out/(ref+'-boundaries.txt')).write_text('\n'.join(f'{t:.6f} {pid}/{tid} {comm} {detail}' for comm,pid,tid,t,detail in relevant)+'\n')
    summary[ref]={'probe_pid':probe,'daemon_pid':daemon,'echo_thread_comm_time':start,'first_daemon_thread_exit_time':end,'window_seconds':end-start,'total_samples':len(samples),'samples':{window+':'+role:n for (window,role),n in grouped.items()},'record_types':dict(collections.Counter(re.split(r'[:(]', r[4].split()[0])[0] for r in records))}
(out/'input-hashes.json').write_text(json.dumps(identities,indent=2)+'\n')
(out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(out/'statuses.json').write_text(json.dumps(statuses,indent=2)+'\n')
shutil.copyfile('/tmp/r23-extract.py',out/'extract.py.txt')
