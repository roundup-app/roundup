from importlib.machinery import SourceFileLoader
import pathlib,subprocess,shlex
m=SourceFileLoader('collect','/tmp/r23-collect.py').load_module()
root=m.ROOT
m.run('profiler-versions',['bash','-c','perf --version; strace --version; cat /proc/sys/kernel/perf_event_paranoid'])
for ref in ['historical','candidate']:
 base=['/tmp/r23-bins/compatible-perf','--runs','1','--calls','50000','--rupd',f'/tmp/r23-bins/{ref}-rupd','--rup',f'/tmp/r23-bins/{ref}-rup','--budgets',str(root/'diagnostic-budgets.json'),'--out',str(root/f'{ref}-profile.json')]
 control=base.copy()
 control[-1]=str(root/f'{ref}-profile-control.json')
 m.run(ref+'-profile-control',['sudo','-n','--']+control,timeout=900)
 m.run(ref+'-profile',['sudo','-n','perf','record','-F','199','-e','cpu-clock','--call-graph','dwarf,16384','-o',f'/tmp/r23-{ref}.data','--']+base,timeout=900)
 m.run(ref+'-retain-raw-profile',['bash','-c',f'sudo -n cat /tmp/r23-{ref}.data | gzip -c > {root}/{ref}-perf.data.gz'])
 m.run(ref+'-cpu-report',['sudo','-n','perf','report','-i',f'/tmp/r23-{ref}.data','--stdio','--show-nr-samples','--percent-limit','0.5','--call-graph','graph,0.5,caller'])
 m.run(ref+'-cpu-flat',['sudo','-n','perf','report','-i',f'/tmp/r23-{ref}.data','--stdio','--show-nr-samples','--percent-limit','0.5','--no-children','--call-graph','none'])
 m.run(ref+'-buildids',['sudo','-n','perf','buildid-list','-i',f'/tmp/r23-{ref}.data'])
 m.run(ref+'-compress-flat',['gzip','-f',str(root/f'{ref}-cpu-flat.stdout')])
 m.run(ref+'-compress-report',['gzip','-f',str(root/f'{ref}-cpu-report.stdout')])
 # Profiled syscalls are a separate diagnostic, never a clean timing pair.
 base[base.index('50000')]='1000'
 base[-1]=str(root/f'{ref}-syscalls.json')
 m.run(ref+'-syscalls',['strace','-f','-c','-w','-o',str(root/f'{ref}-strace-summary.txt')]+base,timeout=900)

# Compress the separate diagnostic reports after retaining observer output/status.
for ref in ['historical','candidate']:
 subprocess.run(['gzip','-f']+[str(root/f'{ref}-{name}.json') for name in ['profile-control','profile','syscalls']],check=True)
