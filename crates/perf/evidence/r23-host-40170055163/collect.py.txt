import pathlib,subprocess,os,json,shlex,time
ROOT=pathlib.Path('/tmp/r23-execution/crates/perf/evidence/r23-host-40170055163')
def run(name,cmd,cwd='/tmp/r23-probe',timeout=900):
 (ROOT/(name+'.command')).write_text('cwd: '+cwd+'\n$ '+shlex.join(cmd)+'\n')
 started=time.perf_counter()
 with (ROOT/(name+'.stdout')).open('w') as out, (ROOT/(name+'.stderr')).open('w') as err:
  try:
   p=subprocess.run(cmd,cwd=cwd,stdout=out,stderr=err,timeout=timeout)
   code=p.returncode
  except subprocess.TimeoutExpired:
   code=124
 (ROOT/(name+'.exit')).write_text(str(code)+'\n')
 (ROOT/(name+'.elapsed-seconds')).write_text(str(time.perf_counter()-started)+'\n')
 print(name,code,flush=True)
 return code

 budgets=json.loads(pathlib.Path('/tmp/r23-candidate/crates/perf/budgets.json').read_text())

 budgets.pop('hook_loop_ms')
 (ROOT/'diagnostic-budgets.json').write_text(json.dumps(budgets,indent=2)+'\n')
 (ROOT/'classes.json').write_text(json.dumps({'linux-9v45':'DIAGNOSTIC ONLY: copied linux-boxd for scratch execution; NOT calibration or a measured baseline; compare the paired historical reports'},indent=2)+'\n')
 for pair in range(1,4):
