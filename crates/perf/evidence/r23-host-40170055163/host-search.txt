$ git show origin/main:.github/workflows/build-issue.yml | head -55
name: build-issue

# L23: one build job and its own `after`, called once per Issue by build.yml's matrix. Each Issue's `after` runs as
# soon as its build ends, so a finished job frees its slot and starts the next pick without waiting for the other
# Issues of its run. `after` holds the write token the Builder never sees.
on:
  workflow_call:
    inputs:
      issue:
        required: true
        type: number
      slug:
        required: true
        type: string

permissions:
  contents: read

# L98: bash -eo pipefail, so a failure left of a pipe fails the step.
defaults:
  run:
    shell: bash

jobs:
  build:
    # A second job for one Issue waits, then finds the first one's PR and starts no run.
    concurrency:
      group: build-${{ inputs.issue }}
      cancel-in-progress: false
    runs-on: ${{ fromJSON(contains(fromJSON(vars.BOXD_ISSUES || '[]'), inputs.issue) && format('["self-hosted","boxd","roundup-build-{0}-{1}-{2}"]', inputs.issue, github.run_id, github.run_attempt) || '["ubuntu-24.04"]') }}
    timeout-minutes: 120
    permissions:
      contents: read
      issues: read
      pull-requests: read
      id-token: write
    outputs:
      ran: ${{ steps.worked.outputs.ran }}
    env:
      ISSUE: ${{ inputs.issue }}
      SLUG: ${{ inputs.slug }}
      # L23: the switch; set, Claude Code runs on API credit (`loop/builder.py` `claude_env`).
      CLAUDE_API: ${{ vars.CLAUDE_API }}
    steps:
      - uses: actions/checkout@v5
        with:
          ref: main
          fetch-depth: 0
          clean: false
          persist-credentials: false
      # The Builder switches this checkout to its branch; the end of the job runs main's copy.
      - run: cp -R loop "$RUNNER_TEMP/loop" && mkdir "$RUNNER_TEMP/.agents" && cp .agents/screen.md "$RUNNER_TEMP/.agents/"
      - uses: ./.github/actions/setup
        with:
          browser: 'true'

exit: 0
