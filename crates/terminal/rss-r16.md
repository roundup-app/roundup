Refs #1326. Allocation repair: [#1329](https://github.com/saadshahd/roundup/pull/1329), merged as `437e4206b7a543e3fd08d66c2f0beb9ec3d3daf5`. ## Reproduction

Create worktrees at the full references in metadata at the recorded paths, build both as above, apply the compatibility patch only to historical perf, and run `cargo build --release -p perf` there (exit 0). Then execute this script (exit 0). It retains all reports even when perf exits 1; check clean qualification and compare RSS medians as described above. The temporary Project, socket and ten login-shell workload are created by the original probe for each Daemon; neither main nor historical source is otherwise edited.

```python
import json,os,pathlib,subprocess,statistics,hashlib,time
root=pathlib.Path('/tmp/rss-evidence')
probe='/tmp/roundup-rss-historical/target/release/perf'
refs={'historical':'/tmp/roundup-rss-historical','repaired':'/tmp/roundup-rss'}
meta={}
for name,cmd in {'os':['cat','/etc/os-release'],'cpu':['lscpu'],'online':['cat','/sys/devices/system/cpu/online'],'available':['nproc'],'affinity':['taskset','-pc',str(os.getpid())],'kernel':['uname','-a'],'rust':['rustc','-Vv'],'cargo':['cargo','-V'],'shell':['bash','--version']}.items():
 meta[name]=subprocess.check_output(cmd,text=True)
meta['env']={k:os.environ.get(k) for k in ['RUSTFLAGS','CARGO_BUILD_RUSTFLAGS','SHELL']}
meta['refs']={n:subprocess.check_output(['git','rev-parse','HEAD'],cwd=p,text=True).strip() for n,p in refs.items()}
meta['hashes']={p:hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest() for p in [probe]+[d+'/target/release/'+b for d in refs.values() for b in ['rupd','rup']]}
(root/'metadata.json').write_text(json.dumps(meta,indent=2))
for pair in range(1,10):
 for name,d in refs.items():
  env=dict(os.environ)
  if name=='repaired':env['R16_CURRENT_PROTOCOL']='1'
  else:env.pop('R16_CURRENT_PROTOCOL',None)
  dest=root/f'paired-{pair}-{name}'
  cmd=[probe,'--rupd',d+'/target/release/rupd','--out',str(dest)+'.json']
  with pathlib.Path(str(dest)+'.log').open('w') as log:
   result=subprocess.run(cmd,cwd=refs['historical'],env=env,stdout=log,stderr=subprocess.STDOUT,timeout=180)
  pathlib.Path(str(dest)+'.exit').write_text(str(result.returncode)+'\n')
  if pathlib.Path(str(dest)+'.json').exists():
   report=json.loads(pathlib.Path(str(dest)+'.json').read_text())
   print(pair,name,result.returncode,report['metrics']['rss_extra_mb'],statistics.median(report['loads'])/report['cpus'],flush=True)
  else: print(pair,name,result.returncode,'NO REPORT',flush=True)

```
