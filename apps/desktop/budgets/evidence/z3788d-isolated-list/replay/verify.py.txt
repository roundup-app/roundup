import pathlib,tempfile,shutil,json,subprocess,hashlib
root=pathlib.Path.cwd();base=root/'apps/desktop/budgets/evidence/z3788d-isolated-list';script=(base/'replay.py.txt').read_text();records=[]
def archive_hashes():return {p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in base.iterdir() if p.is_file()}
before=archive_hashes()
for name in ['missing-sample','wrong-identities','missing-presentation','wrong-trace-pair']:
 with tempfile.TemporaryDirectory(prefix='z3788d-replay-') as d:
  archive=pathlib.Path(d)
  for path in base.iterdir():
   if path.is_file(): (archive/path.name).symlink_to(path)
  if name in ['missing-sample','wrong-identities']:
   file=archive/'40-samples.json';rows=json.loads(file.read_text());file.unlink()
   if name=='missing-sample':rows.pop()
   else:rows[3]['identitiesSha256']='wrong'
   file.write_text(json.dumps(rows))
  source=script.replace("archive=out;out=archive/'replay'",f"archive=pathlib.Path({str(archive)!r});out=archive/'replay'")
  if name=='missing-presentation':source=source.replace("for e in sorted(stages,key=lambda event:event['ts']):","for e in []:")
  if name=='wrong-trace-pair':source=source.replace("if hashlib.sha256(raw).hexdigest()!=expected_hash:","if hashlib.sha256(raw).hexdigest()!='wrong':")
  run=subprocess.run(['python3','-c',source],cwd=root,capture_output=True,text=True)
  result=json.loads((archive/'replay/result.json').read_text())
  assert run.returncode==2 and result['exit']==2 and result['failures'],(name,run.returncode,run.stderr)
  records.append({'case':name,'exit':run.returncode,'failures':result['failures']})
result=json.loads((base/'replay/result.json').read_text())
assert result['exit']==1 and len(result['reports'])==3 and not result['failures'] and not result['selectionFailures']
records.insert(0,{'case':'all archived samples','exit':1,'validated':660,'budgetPass':False})
assert before==archive_hashes(),'Original archive changed'
(base/'replay/validation.json').write_text(json.dumps({'replaySha256':hashlib.sha256(script.encode()).hexdigest(),'checks':records,'originalArchivesUnchanged':True},indent=2)+'\n')
print(json.dumps(records,indent=2))
