 config=pathlib.Path(temp)/'config.json';config.write_text('{}')
 def browser(*args):
  p=subprocess.run([str(root/'apps/desktop/node_modules/.bin/agent-browser'),'--session',session,'--profile',temp+'/profile','--config',str(config),'--executable-path','/opt/google/chrome/chrome','--args','--no-sandbox','--json',*args],capture_output=True,text=True,timeout=50)
  if p.returncode:raise RuntimeError(p.stdout+p.stderr)
  r=json.loads(p.stdout)
  if not r.get('success'):raise RuntimeError(r)
  return r['data']
 def ev(code):return browser('eval',code)['result']
