import { it, expect } from "vitest";
import { openLane } from "../todos/laneBrowser";
import { startRealDaemon } from "../testing/realDaemonFixture";
import { createServedApp } from "../testing/realSeam";
import { mkdirSync, writeFileSync } from "node:fs";
import { resolve } from "node:path";

it("z3788a_real_daemon_serialized_reload_observer", async () => {
  const real = await startRealDaemon({}, true);
  const page = await openLane(1512, "dark", "z3788a-serialized-reload");
  const dir = resolve("../../artifacts/ux/Z3788a");
  mkdirSync(dir, { recursive: true });

  try {
    const app = createServedApp(real.base);
    const home = await app.rpc("rail.createWorkstream", { name: "Scrollback", parent: null });
    const shell = await app.rpc("terminal.open", { workstream: home.id });
    await page.browser("open", `${real.base}/harness.html?view=workstreams&items=1&tabs=1&daemon=1&renderer=dom`);
    await page.until(`!!document.querySelector('.rail-row[data-id="${home.id}"]')`);
    await page.browser("focus", `.rail-row[data-id="${home.id}"]`);
    await page.browser("press", "Enter");
    await page.until(`document.querySelector('.rail-row[data-id="${home.id}"]')?.dataset.selected === 'true'`);
    await page.until("!!document.querySelector('.terminal-tab')");
    await page.browser("click", `.terminal-tab[data-bubble="${shell.terminal_id}"]`);
    await page.until(`!!window.__terminals?.get('${shell.terminal_id}')`);
    await app.rpc("terminal.write", { id: shell.terminal_id, data: Buffer.from("printf 'line %s\\n' {1..100}\nprintf '\\033[1;31mwide 世界 styled\\033[0m\\n'\n").toString("base64") });
    await page.until(`Object.keys(localStorage).some(key => key.startsWith('roundup:terminal-scrollback:') && JSON.parse(localStorage.getItem(key)).data.includes('line 100'))`);
    await page.capture("Z3788a", "serialized-before-reload");
    const saved = await page.evaluate<string>(`window.__terminals.get('${shell.terminal_id}').serialize().data`);
    expect(saved).toContain("line 1\r\n");
    await page.browser("reload");
    await page.until(`!!window.__terminals?.get('${shell.terminal_id}')`);
    await page.until(`window.__terminals.get('${shell.terminal_id}').serialize().data.includes('line 100')`);
    expect(await page.evaluate<string>(`window.__terminals.get('${shell.terminal_id}').serialize().data`)).toContain("line 1\r\n");
    await page.capture("Z3788a", "serialized-after-reload");
    const checks = await page.evaluate<Record<string, {status:string}>>("window.__checks()");
    writeFileSync(resolve(dir, "serialized-reload-observation.json"), JSON.stringify({ checks, realDaemon: true, renderer: "dom" }, null, 2));
  } finally { await page.close(); await real.stop(); }
}, 300_000);
